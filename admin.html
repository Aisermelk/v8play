import { db } from "./firebase.js"; import { doc, updateDoc } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-firestore.js"; // Função para salvar a alteração do título async function salvarTituloItem(colecao, idItem, novoTitulo) { if (!novoTitulo || !idItem || !colecao) { alert("Preencha o título e certifique-se de que o item é válido."); return; } try { // Aponta para o documento específico na coleção correta (filmes, series, animes, doramas) const itemRef = doc(db, colecao, idItem); // Atualiza apenas o campo 'titulo' no Firestore await updateDoc(itemRef, { titulo: novoTitulo }); alert("Título atualizado com sucesso!"); // Recarrega a listagem do admin para refletir a mudança if (typeof carregarCatalogoAdmin === "function") { carregarCatalogoAdmin(); } } catch (error) { console.error("Erro ao atualizar o título no Firestore:", error); alert("Erro ao salvar alterações. Verifique o console ou as regras do Firebase."); } }
